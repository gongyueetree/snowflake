cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico)
set(PICO_DEFAULT_BOOT_STAGE2 boot2_generic_03h)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(snowflake_player C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(snowflake_player main.c)
target_compile_definitions(snowflake_player PRIVATE PICO_FLASH_SIZE_BYTES=8388608)
target_link_libraries(snowflake_player pico_stdlib hardware_pio hardware_clocks)
pico_enable_stdio_usb(snowflake_player 0)
pico_enable_stdio_uart(snowflake_player 0)
pico_generate_pio_header(snowflake_player ${CMAKE_CURRENT_LIST_DIR}/ws2812.pio)
add_custom_command(TARGET snowflake_player POST_BUILD COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:snowflake_player> ${CMAKE_CURRENT_BINARY_DIR}/runtime.bin)
